## 消费的兄弟合同（Phase 1.2 已对齐的形状，假件照此编）

- **arbiter**（`api:acquireHeavy` / `api:releaseHeavy` / `api:reapLlmPort` / `data:deskState`）：
  - `acquire_heavy(kind, label) -> {"ok": True, "token": str, ...} | {"ok": False, "reason": {"code", "message"}}`
  - `release_heavy(token) -> {"ok": bool, ...}`（token 失效 → `not_holder`，无副作用）
  - `reap_llm_port(port) -> {"ok": bool, "port": int, "killed_pids": [int], "error": str|None}`
  - `desk_state() -> {"holder": None | {"kind","label","phase",...}, "media_busy": bool, ...}`
  - `event:heavyStateChanged` 载荷即当刻 deskState；组装层把 `LlmService.on_heavy_state_changed`
    订阅进去（订阅动作归组装层，本模块只提供回调方法）。
- **resources**（`api:listCatalog` / `data:modelEntry`）：条目属性
  `key/name/group/hf_repo/relpath/gb/vision/quant/params`。
- **foundation**（`api:resolvePaths`）：本模块只用 `models_root`、`logs_dir`、`venv_python` 三个字段。

```python
# desk/llm/state.py
"""llm 模块的数据形状与错误码（data:llmState / data:loadedModel / ChatEvent）。"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Any

# mlx-lm 内部端口的仓库唯一定义点（冻结约定；组装层把它传给 Arbiter(llm_port=...)）。
DEFAULT_LLM_PORT = 8767

STATUS_IDLE = "idle"
STATUS_LOADING = "loading"
STATUS_LOADED = "loaded"
STATUS_ERROR = "error"

ERR_MODEL_NOT_FOUND = "model_not_found"
ERR_MODEL_DIR_MISSING = "model_dir_missing"
ERR_MEDIA_BUSY = "media_busy"
ERR_LOAD_IN_PROGRESS = "load_in_progress"
ERR_BACKEND_EXITED = "backend_exited"
ERR_LOAD_TIMEOUT = "load_timeout"
ERR_PORT_NOT_RELEASED = "port_not_released"
ERR_EVICTED = "evicted"
ERR_NO_MODEL_LOADED = "no_model_loaded"
ERR_MODEL_MISMATCH = "model_mismatch"
ERR_UPSTREAM_ERROR = "upstream_error"


class LlmRejected(Exception):
    """拒绝（非故障）：不改写状态机，由 HTTP 层映射 4xx/503（设计 Assumption 5）。"""

    def __init__(self, code: str, message: str):
        super().__init__(message)
        self.code = code
        self.message = message

    def to_dict(self) -> dict[str, Any]:
        return {"code": self.code, "message": self.message}


class UpstreamError(Exception):
    """对话中的上游故障（非流式 502 / 流内 error 事件）。不补造数值。"""

    def __init__(self, message: str):
        super().__init__(message)
        self.code = ERR_UPSTREAM_ERROR
        self.message = message


@dataclass(frozen=True)
class LlmError:
    code: str
    message: str
    log_tail: str | None = None

    def to_dict(self) -> dict[str, Any]:
        return {"code": self.code, "message": self.message, "log_tail": self.log_tail}


@dataclass(frozen=True)
class LlmState:
    """data:llmState"""
    status: str = STATUS_IDLE
    model_key: str | None = None
    error: LlmError | None = None
    loaded_at: float | None = None

    def to_dict(self) -> dict[str, Any]:
        return {
            "status": self.status,
            "model_key": self.model_key,
            "error": self.error.to_dict() if self.error else None,
            "loaded_at": self.loaded_at,
        }


@dataclass(frozen=True)
class LoadedModel:
    """data:loadedModel —— 字段零拷贝自 resources 的 data:modelEntry（R-llm-05）。"""
    key: str
    name: str
    hf_repo: str
    served_id: str
    vision: bool
    quant: str | None
    params: str | None
    gb: float

    @classmethod
    def from_entry(cls, entry: Any) -> "LoadedModel":
        return cls(key=entry.key, name=entry.name, hf_repo=entry.hf_repo,
                   served_id=entry.hf_repo, vision=bool(entry.vision),
                   quant=entry.quant, params=entry.params, gb=float(entry.gb))

    def to_dict(self) -> dict[str, Any]:
        return {"key": self.key, "name": self.name, "hf_repo": self.hf_repo,
                "served_id": self.served_id, "vision": self.vision,
                "quant": self.quant, "params": self.params, "gb": self.gb}


def delta_event(text: str | None, reasoning: str | None) -> dict[str, Any]:
    """二者至少一个非空；正文与思考永不合并（R-llm-03）。"""
    return {"type": "delta", "text": text, "reasoning": reasoning}


def done_event(usage: dict, finish_reason: str) -> dict[str, Any]:
    """成功终止必带 usage 与 finish_reason（与 gateway 的冻结合同）。"""
    return {"type": "done", "usage": usage, "finish_reason": finish_reason}


def error_event(code: str, message: str) -> dict[str, Any]:
    return {"type": "error", "code": code, "message": message}
```

```python
# tests/llm/llm_fakes.py
"""llm 测试共用假件：可编程脚本（退出码序列、health 序列、chunk 序列）+ 共享调用台账。"""
from __future__ import annotations

import threading
from dataclasses import dataclass
from pathlib import Path
from types import SimpleNamespace

from desk.llm.backend import BackendHttpError


@dataclass(frozen=True)
class FakeEntry:
    key: str
    name: str
    group: str
    hf_repo: str
    relpath: str
    gb: float
    vision: bool = False
    quant: str | None = None
    params: str | None = None


def make_entry(key: str = "glm", **kw) -> FakeEntry:
    defaults = dict(name="GLM-4.5-Air", group="chat",
                    hf_repo=f"mlx-community/{key}-4bit",
                    relpath=f"llms/mlx-community/{key}-4bit",
                    gb=60.2, vision=False, quant="4bit", params="106B")
    defaults.update(kw)
    return FakeEntry(key=key, **defaults)


def _pop(script: list):
    if len(script) > 1:
        return script.pop(0)
    return script[0]


class FakeCatalog:
    def __init__(self, entries):
        self._entries = list(entries)

    def list_catalog(self):
        return list(self._entries)


class FakePaths:
    """resolve_paths() 只需给出本模块消费的三个字段。"""

    def __init__(self, root: Path):
        self.models_root = root / "models"
        self.logs_dir = root / "logs"
        self.venv_python = root / "venv" / "bin" / "python3"
        self.models_root.mkdir(parents=True, exist_ok=True)
        self.logs_dir.mkdir(parents=True, exist_ok=True)

    def resolve_paths(self):
        return SimpleNamespace(models_root=self.models_root,
                               logs_dir=self.logs_dir,
                               venv_python=self.venv_python)


class FakeProcess:
    def __init__(self, poll_script=(None,), calls: list | None = None):
        self._script = list(poll_script)
        self.calls = calls if calls is not None else []

    def set_exit(self, code: int) -> None:
        self._script = [code]

    def poll(self):
        return _pop(self._script)

    def terminate(self):
        self.calls.append(("terminate",))

    def kill(self):
        self.calls.append(("kill",))

    def wait(self, timeout):
        self.calls.append(("wait", timeout))


class FakeBackend:
    def __init__(self, *, process: FakeProcess | None = None, health_script=(True,),
                 chunks=(), stream_error_after: int | None = None,
                 chat_result: dict | None = None, chat_error: Exception | None = None,
                 tail_text: str = "fake mlx-lm log tail", calls: list | None = None):
        self.calls = calls if calls is not None else []
        self.process = process if process is not None else FakeProcess(calls=self.calls)
        self._health_script = list(health_script)
        self.chunks = list(chunks)
        self.stream_error_after = stream_error_after
        self.chat_result = chat_result
        self.chat_error = chat_error
        self.tail_text = tail_text
        self.spawn_exc: Exception | None = None
        self.hold_health = False
        self.health_release = threading.Event()

    def spawn(self, python, model_path, port, log_path):
        self.calls.append(("spawn", str(python), str(model_path), port, str(log_path)))
        if self.spawn_exc is not None:
            raise self.spawn_exc
        return self.process

    def health(self, port):
        self.calls.append(("health", port))
        if self.hold_health:
            return bool(self.health_release.wait(timeout=5.0))
        return _pop(self._health_script)

    def chat(self, port, payload):
        self.calls.append(("chat", port, payload))
        if self.chat_error is not None:
            raise self.chat_error
        return self.chat_result

    def chat_stream(self, port, payload):
        self.calls.append(("chat_stream", port, payload))

        def gen():
            for i, chunk in enumerate(self.chunks):
                if self.stream_error_after is not None and i == self.stream_error_after:
                    raise BackendHttpError("fake 上游流中断")
                yield chunk
            if self.stream_error_after is not None and self.stream_error_after >= len(self.chunks):
                raise BackendHttpError("fake 上游流中断")

        return gen()

    def log_tail(self, log_path, max_lines: int = 40):
        self.calls.append(("log_tail", str(log_path)))
        return self.tail_text


class FakeArbiter:
    def __init__(self, *, calls: list | None = None, acquire_results=None,
                 reap_results=None, desk_states=None):
        self.calls = calls if calls is not None else []
        self._acquire = list(acquire_results or [{"ok": True, "token": "tok-1"}])
        self._reap = list(reap_results or
                          [{"ok": True, "port": 0, "killed_pids": [], "error": None}])
        self._desk_states = list(desk_states or [{
            "holder": {"kind": "llm", "label": "glm", "phase": "held"},
            "media_busy": False}])

    def acquire_heavy(self, kind, label):
        self.calls.append(("acquire", kind, label))
        return _pop(self._acquire)

    def release_heavy(self, token):
        self.calls.append(("release", token))
        return {"ok": True}

    def reap_llm_port(self, port):
        self.calls.append(("reap", port))
        return _pop(self._reap)

    def desk_state(self):
        # 故意不记台账：desk_state 是高频只读轮询，不参与顺序断言。
        return _pop(self._desk_states)

    def set_desk_state(self, state: dict) -> None:
        self._desk_states = [state]
```

```python
"""MlxLmBackend.spawn 的 argv 契约与 log_tail（R-llm-01/04；packaging §1.3 的 -s 约定）。"""
import subprocess
from pathlib import Path
from types import SimpleNamespace

from desk.llm.backend import MlxLmBackend


class PopenRecorder:
    def __init__(self):
        self.argv = None
        self.kwargs = None

    def __call__(self, argv, **kwargs):
        self.argv = list(argv)
        self.kwargs = kwargs
        return SimpleNamespace(poll=lambda: None)


def test_spawn_argv_exact_no_allowed_origins(tmp_path):
    rec = PopenRecorder()
    b = MlxLmBackend(popen=rec)
    log = tmp_path / "logs" / "mlx-lm.log"
    b.spawn(Path("/venv/bin/python3"), tmp_path / "model", 18767, log)
    assert rec.argv == ["/venv/bin/python3", "-s", "-m", "mlx_lm", "server",
                       "--model", str(tmp_path / "model"),
                       "--host", "127.0.0.1", "--port", "18767"]
    # 浏览器不再直连 mlx-lm：绝不放开 CORS（设计 Assumption 6）
    assert not any("allowed-origins" in a for a in rec.argv)
    assert rec.kwargs["stderr"] is subprocess.STDOUT
    assert log.parent.is_dir()          # 日志目录已建
    # stdout 以追加模式指向 log_path
    assert rec.kwargs["stdout"].name == str(log)


def test_log_tail_returns_last_40_lines(tmp_path):
    log = tmp_path / "mlx-lm.log"
    log.write_text("\n".join(f"line-{i}" for i in range(60)) + "\n")
    tail = MlxLmBackend().log_tail(log)
    lines = tail.splitlines()
    assert len(lines) == 40 and lines[0] == "line-20" and lines[-1] == "line-59"


def test_log_tail_missing_file_placeholder_not_exception(tmp_path):
    missing = tmp_path / "nope.log"
    tail = MlxLmBackend().log_tail(missing)
```

```python
"""LlmBackend 协议 + MlxLmBackend 真实实现（subprocess + http.client）。唯一可替换点。"""
from __future__ import annotations

import subprocess
from collections.abc import Iterator
from pathlib import Path
from typing import Protocol


class BackendHttpError(Exception):
    """mlx-lm HTTP 层故障（连接拒绝、非 200、流中断）——必须抛出，不吞。"""


class BackendProcess(Protocol):
    def poll(self) -> int | None: ...
    def terminate(self) -> None: ...
    def kill(self) -> None: ...
    def wait(self, timeout: float) -> None: ...


class LlmBackend(Protocol):
    def spawn(self, python: Path, model_path: Path, port: int, log_path: Path) -> BackendProcess: ...
    def health(self, port: int) -> bool: ...
    def chat(self, port: int, payload: dict) -> dict: ...
    def chat_stream(self, port: int, payload: dict) -> Iterator[dict]: ...
    def log_tail(self, log_path: Path, max_lines: int = 40) -> str: ...


class MlxLmBackend:
    def __init__(self, popen=subprocess.Popen):
        self._popen = popen

    def spawn(self, python: Path, model_path: Path, port: int, log_path: Path):
        log_path.parent.mkdir(parents=True, exist_ok=True)
        argv = [str(python), "-s", "-m", "mlx_lm", "server",
                "--model", str(model_path), "--host", "127.0.0.1", "--port", str(port)]
        log_file = open(log_path, "ab")
        try:
            return self._popen(argv, stdout=log_file, stderr=subprocess.STDOUT)
        finally:
            log_file.close()   # 子进程已继承 fd

    def log_tail(self, log_path: Path, max_lines: int = 40) -> str:
        try:
            text = log_path.read_text(encoding="utf-8", errors="replace")
        except OSError:
            return f"日志文件不存在: {log_path}"
        return "\n".join(text.splitlines()[-max_lines:])
```

```python
        try:

                         headers={"Content-Type": "application/json"})
            resp = conn.getresponse()
        except OSError as exc:
            conn.close()
            raise BackendHttpError(f"连接 mlx-lm 失败: {exc}") from exc
        if resp.status != 200:
            head = resp.read(2048)
            conn.close()

        try:
            while True:
                try:
                    raw = resp.readline()
                except OSError as exc:
                    raise BackendHttpError(f"上游流中断: {exc}") from exc

                    return                      # 流终止（有无 [DONE] 都停；合同判定归 service）

                if data == "[DONE]":
                    return

                    yield json.loads(data)
```

```python
# desk/llm/service.py （T-llm-05 初版）
"""LlmService：加载/卸载/状态/对话编排。互斥判断完全不做——一律问 arbiter。"""
from __future__ import annotations

import threading
import time
from pathlib import Path
from typing import Any

from desk.llm.state import (
    DEFAULT_LLM_PORT,
    STATUS_ERROR, STATUS_IDLE, STATUS_LOADED, STATUS_LOADING,
    ERR_BACKEND_EXITED, ERR_EVICTED, ERR_LOAD_IN_PROGRESS, ERR_LOAD_TIMEOUT,
    ERR_MEDIA_BUSY, ERR_MODEL_DIR_MISSING, ERR_MODEL_MISMATCH, ERR_MODEL_NOT_FOUND,
    ERR_NO_MODEL_LOADED, ERR_PORT_NOT_RELEASED,
    LlmError, LlmRejected, LlmState, LoadedModel,
)


class LlmService:
    def __init__(self, backend, arbiter, catalog, paths, *,
                 port: int = DEFAULT_LLM_PORT,
                 load_timeout_s: float = 180.0,
                 poll_interval_s: float = 0.4,
                 term_grace_s: float = 8.0):
        self._backend = backend
        self._arbiter = arbiter
        self._catalog = catalog
        self._paths = paths
        self._port = port
        self._load_timeout_s = load_timeout_s
        self._poll_interval_s = poll_interval_s
        self._term_grace_s = term_grace_s
        self._lock = threading.Lock()
        self._state = LlmState()
        self._proc = None
        self._token: str | None = None
        self._entry = None            # 驻留条目：data:loadedModel 的来源
        self._load_thread: threading.Thread | None = None

    # ---------- 内部小件 ----------
    def _find_entry(self, key: str):
        for entry in self._catalog.list_catalog():
            if entry.key == key:
                return entry
        return None

    @staticmethod
    def _log_path(roots) -> Path:
        return Path(roots.logs_dir) / "mlx-lm.log"

    # ---------- api:loadLlm ----------
    def load(self, key: str) -> dict[str, Any]:
        with self._lock:
            entry = self._find_entry(key)
            roots = self._paths.resolve_paths()
            model_dir = Path(roots.models_root) / entry.relpath
            res = self._arbiter.acquire_heavy("llm", entry.key)
            self._token = res.get("token")
            self._entry = None
            self._state = LlmState(status=STATUS_LOADING, model_key=entry.key)
            thread = threading.Thread(
                target=self._load_worker,
                args=(entry, model_dir, Path(roots.venv_python), self._log_path(roots)),
                daemon=True, name="llm-load")
            self._load_thread = thread
            thread.start()
            return self._state.to_dict()

    def _load_worker(self, entry, model_dir: Path, python: Path, log_path: Path) -> None:
        self._arbiter.reap_llm_port(self._port)     # spawn 前先收割孤儿（A06/A07）
        proc = self._backend.spawn(python, model_dir, self._port, log_path)
        with self._lock:
            self._proc = proc
        deadline = time.monotonic() + self._load_timeout_s
        while True:
            rc = proc.poll()
            if rc is not None:
                with self._lock:
                    self._proc = None
                    self._state = LlmState(status=STATUS_ERROR, model_key=entry.key,
                                           error=LlmError(ERR_BACKEND_EXITED,
                                                          f"mlx-lm 进程退出，退出码 {rc}"))
                return
            if self._backend.health(self._port):
                with self._lock:
                    self._state = LlmState(status=STATUS_LOADED, model_key=entry.key,
                                           loaded_at=time.time())
                    self._entry = entry
                return
            if time.monotonic() >= deadline:
                with self._lock:
                    self._proc = None
                    self._state = LlmState(status=STATUS_ERROR, model_key=entry.key,
                                           error=LlmError(ERR_LOAD_TIMEOUT,
                                                          f"mlx-lm 超过 {self._load_timeout_s:.0f}s 未就绪"))
                return
            time.sleep(self._poll_interval_s)

    # ---------- api:llmStatus（基础版；存活自检在 T-llm-10 补） ----------
    def status(self) -> dict[str, Any]:
        with self._lock:
            loaded = None
            if self._state.status == STATUS_LOADED and self._entry is not None:
                loaded = LoadedModel.from_entry(self._entry).to_dict()
            return {"state": self._state.to_dict(), "loaded_model": loaded}

    # ---------- 辅助：等状态离开 loading（测试与调用方轮询用） ----------
    def wait_settled(self, timeout_s: float = 5.0) -> dict[str, Any]:
        deadline = time.monotonic() + timeout_s
        while time.monotonic() < deadline:
            with self._lock:
                if self._state.status != STATUS_LOADING:
                    return self._state.to_dict()
            time.sleep(0.001)
        with self._lock:
            return self._state.to_dict()
```

```python
        if not reap.get("ok"):
            with self._lock:
                self._fail_locked(ERR_PORT_NOT_RELEASED,
                                  f"加载前收割后端口 {self._port} 仍被占用: {reap.get('error')}",
                                  log_tail=None, release=True)
            return
        try:
            proc = self._backend.spawn(python, model_dir, self._port, log_path)
        except Exception as exc:
            with self._lock:
                self._fail_locked(ERR_BACKEND_EXITED, f"mlx-lm 启动失败: {exc}",
                                  log_tail=self._backend.log_tail(log_path), release=True)
            return
        with self._lock:
            self._proc = proc
        deadline = time.monotonic() + self._load_timeout_s
        while True:
            rc = proc.poll()
            if rc is not None:
                with self._lock:
                    self._proc = None
                    self._fail_locked(ERR_BACKEND_EXITED, f"mlx-lm 进程退出，退出码 {rc}",
                                      log_tail=self._backend.log_tail(log_path), release=True)
                return
            if self._backend.health(self._port):
                with self._lock:
                    self._state = LlmState(status=STATUS_LOADED, model_key=entry.key,
                                           loaded_at=time.time())
                    self._entry = entry
                return
            if time.monotonic() >= deadline:
                proc.kill()
                self._arbiter.reap_llm_port(self._port)          # 超时清尾
                with self._lock:
                    self._proc = None
                    self._fail_locked(ERR_LOAD_TIMEOUT,
                                      f"mlx-lm 超过 {self._load_timeout_s:.0f}s 未就绪",
                                      log_tail=self._backend.log_tail(log_path), release=True)
                return
            time.sleep(self._poll_interval_s)
```

```python
    # ---------- 对话前置检查（两个 API 相同；同步、立即，绝不等待——R-llm-06） ----------
    def _chat_precheck(self, request: dict[str, Any]):
        ds = self._arbiter.desk_state() or {}
        if ds.get("media_busy"):
            raise LlmRejected(ERR_MEDIA_BUSY, "媒体作业进行中，聊天请求被拒绝")
        with self._lock:
            if self._state.status != STATUS_LOADED or self._entry is None:
                raise LlmRejected(ERR_NO_MODEL_LOADED, "当前没有加载模型")
            entry = self._entry
        req_model = request.get("model")
        if req_model and req_model not in (entry.key, entry.hf_repo):
            raise LlmRejected(ERR_MODEL_MISMATCH,
                              f"请求模型 {req_model!r} 与驻留模型 {entry.key!r} 不符")
        return entry

    @staticmethod
    def _upstream_payload(request: dict[str, Any], entry) -> dict[str, Any]:
        payload: dict[str, Any] = {"model": entry.hf_repo, "messages": request["messages"]}
        for k in ("temperature", "top_p", "max_tokens"):
            if request.get(k) is not None:
                payload[k] = request[k]
        return payload

    # ---------- api:chatCompletion ----------
    def chat_completion(self, request: dict[str, Any]) -> dict[str, Any]:
        entry = self._chat_precheck(request)
        payload = self._upstream_payload(request, entry)
        payload["stream"] = False
        try:
            raw = self._backend.chat(self._port, payload)
        except BackendHttpError as exc:
            raise UpstreamError(str(exc)) from exc
        try:
            choice = raw["choices"][0]
            message = choice["message"]
        except (KeyError, IndexError, TypeError) as exc:
            raise UpstreamError(f"上游响应缺 choices/message: {exc}") from exc
        usage = raw.get("usage")
        finish = choice.get("finish_reason")
        if usage is None or finish is None:
            raise UpstreamError("上游响应缺 usage/finish_reason")       # 不估算、不补造
        reasoning = message.get("reasoning_content")
        if reasoning is None:
            reasoning = message.get("reasoning")
        content = message.get("content")
        return {"content": content if content is not None else "",
                "reasoning": reasoning,
                "usage": usage, "finish_reason": finish, "model": entry.key}
```

```python
def serve(routes):
    """把 Route 表挂进一个微型 HTTP 服务——只为证明状态码与帧格式，无业务逻辑。"""
    table = {(r.method, r.path): r.handler for r in routes}

    class H(BaseHTTPRequestHandler):
        def log_message(self, *a):
            pass

        def _dispatch(self):
            handler = table[(self.command, self.path)]
            length = int(self.headers.get("Content-Length") or 0)
            body = json.loads(self.rfile.read(length) or b"{}") if length else {}
            result = handler(body)
            if result.sse is not None:
                self.send_response(result.status)
                self.send_header("Content-Type", "text/event-stream")
                self.end_headers()
                for event in result.sse:
                    self.wfile.write(encode_sse(event))
                return                                        # done 后关流
            payload = json.dumps(result.body).encode()
            self.send_response(result.status)
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(payload)))
            self.end_headers()
            self.wfile.write(payload)

        do_GET = _dispatch
        do_POST = _dispatch

    server = ThreadingHTTPServer(("127.0.0.1", 0), H)
    threading.Thread(target=server.serve_forever, daemon=True).start()
```

```python
# desk/llm/routes.py
"""台面 HTTP 适配：HTTP ↔ LlmService 翻译，零业务逻辑。由服务组装层注册进 8766。"""
from __future__ import annotations

import json
from dataclasses import dataclass
from typing import Any, Callable, Iterator

from desk.llm.state import (
    ERR_LOAD_IN_PROGRESS, ERR_MEDIA_BUSY, ERR_MODEL_DIR_MISSING, ERR_MODEL_MISMATCH,
    ERR_MODEL_NOT_FOUND, ERR_NO_MODEL_LOADED, ERR_UPSTREAM_ERROR,
    LlmRejected, UpstreamError,
)

_REJECTION_STATUS = {
    ERR_MODEL_NOT_FOUND: 404,
    ERR_MODEL_DIR_MISSING: 404,
    ERR_MEDIA_BUSY: 409,
    ERR_LOAD_IN_PROGRESS: 409,
    ERR_MODEL_MISMATCH: 409,
    ERR_NO_MODEL_LOADED: 503,
}


@dataclass(frozen=True)
class RouteResult:
    status: int
    body: dict[str, Any] | None = None
    sse: Iterator[dict] | None = None     # 非空则以 SSE 帧发出，done/error 后关流


@dataclass(frozen=True)
class Route:
    method: str
    path: str
    handler: Callable[[dict], RouteResult]


def encode_sse(event: dict) -> bytes:
    """每个 ChatEvent 一帧：data: {...}\\n\\n"""
    return b"data: " + json.dumps(event, ensure_ascii=False).encode("utf-8") + b"\n\n"


def build_routes(service) -> list[Route]:
    return [
        Route("POST", "/api/llm/load", lambda body: _load(service, body)),
        Route("POST", "/api/llm/unload", lambda body: _unload(service, body)),
        Route("GET", "/api/llm/status", lambda body: RouteResult(200, service.status())),
        Route("POST", "/api/llm/chat", lambda body: _chat(service, body)),
        Route("POST", "/api/llm/chat/stream", lambda body: _chat_stream(service, body)),
    ]


def _rejected(exc: LlmRejected) -> RouteResult:
    return RouteResult(_REJECTION_STATUS.get(exc.code, 409), {"error": exc.to_dict()})


def _load(service, body: dict) -> RouteResult:
    try:
        state = service.load(str(body.get("key") or ""))
    except LlmRejected as exc:
        return _rejected(exc)
    if state["status"] == "error":        # 换模型收尾失败等：500 经状态体表达
        return RouteResult(500, {"state": state, "error": state["error"]})
    return RouteResult(202, {"state": state})


def _unload(service, body: dict) -> RouteResult:
    try:
        state = service.unload()
    except LlmRejected as exc:
        return _rejected(exc)
    if state["status"] == "error":
        return RouteResult(500, {"state": state, "error": state["error"]})
    return RouteResult(200, {"state": state})


def _chat(service, body: dict) -> RouteResult:
    try:
        return RouteResult(200, service.chat_completion(body))
    except LlmRejected as exc:
        return _rejected(exc)
    except UpstreamError as exc:
        return RouteResult(502, {"error": {"code": ERR_UPSTREAM_ERROR,
                                           "message": exc.message}})


def _chat_stream(service, body: dict) -> RouteResult:
    try:
        events = service.chat_stream(body)  # 前置拒绝在发头之前抛出
    except LlmRejected as exc:
        return _rejected(exc)
    return RouteResult(200, sse=events)
```
