```python
"""剧本与闸门：e2e 假件的确定性控制原语（e2e 设计 §2.4）。

纯 stdlib；不 import 任何 desk. 兄弟包。剧本饿死表现为 ScriptStall（清晰失败），
步序耗尽表现为 ScriptExhausted（编程错误即错误，不静默补帧）。
"""
from __future__ import annotations

import threading
from collections import deque
from dataclasses import dataclass
from pathlib import Path

DEFAULT_GATE_TIMEOUT_S = 5.0


class ScriptStall(RuntimeError):
    """闸门在超时内没被放行——剧本饿死，不挂死。"""


class ScriptExhausted(RuntimeError):
    """剧本步序耗尽仍被拉取/放行——测试编排错误。"""


class Gate:
    def __init__(self, timeout_s: float = DEFAULT_GATE_TIMEOUT_S):
        self._event = threading.Event()
        self._timeout_s = timeout_s

    def open(self) -> None:
        self._event.set()

    def wait(self) -> None:
        if not self._event.wait(self._timeout_s):
            raise ScriptStall(f"gate not opened within {self._timeout_s}s")


@dataclass(frozen=True)
class Delta:
    content: str | None = None
    reasoning: str | None = None


@dataclass(frozen=True)
class Done:
    usage: dict
    finish_reason: str = "stop"


@dataclass(frozen=True)
class Line:
    text: str


class _WriteOutput:
    """媒体剧本步：把极小真产物写到 argv 解析出的 --output 路径。"""


WRITE_OUTPUT = _WriteOutput()


@dataclass(frozen=True)
class Exit:
    code: int = 0


class _BlockUntilCancel:
    """媒体剧本步：阻塞直到 handle 收到 terminate/kill（R-e2e-08 的运行中态）。"""


BLOCK_UNTIL_CANCEL = _BlockUntilCancel()


def _gates(steps) -> list[Gate]:
    return [s for s in steps if isinstance(s, Gate)]


def default_chat_steps() -> list:
    return [
        Delta(reasoning="让我想想，"), Gate(),
        Delta(reasoning="想好了。"), Gate(),
        Delta(content="你好"), Gate(),
        Delta(content="，世界"),
        Done(usage={"prompt_tokens": 5, "completion_tokens": 4, "total_tokens": 9}),
    ]


class ChatScript:
    """FakeLlmBackend 的一次性流式剧本。step() 按序放行下一道闸。"""

    def __init__(self, steps: list | None = None):
        self.steps = steps if steps is not None else default_chat_steps()
        self._opened = 0
        self.consumed = False
        self.requests: list[dict] = []   # FakeLlmBackend 收到的 payload（§5.1 反向断言面）

    def step(self) -> None:
        gates = _gates(self.steps)
        if self._opened >= len(gates):
            raise ScriptExhausted("no more gates in chat script")
        gates[self._opened].open()
        self._opened += 1

    def run_to_end(self) -> None:
        for gate in _gates(self.steps)[self._opened:]:
            gate.open()
        self._opened = len(_gates(self.steps))


def default_media_steps() -> list:
    return [Line("step 1/10"), Gate(), Line("step 5/10"), Gate(), WRITE_OUTPUT, Exit(0)]


def fast_media_steps() -> list:
    """不闸停的快进作业（library/回填测试用）。"""
    return [Line("step 1/10"), Line("step 10/10"), WRITE_OUTPUT, Exit(0)]


def cancellable_media_steps() -> list:
    """阻塞直到被取消的作业（互斥/取消场景用）。"""
    return [Line("step 1/10"), Gate(), BLOCK_UNTIL_CANCEL]


class MediaScript:
    """FakeMediaExecutor 的作业队列剧本；每次 spawn 弹出一单。"""

    def __init__(self, jobs: list[list] | None = None):
        self.jobs: deque[list] = deque(jobs if jobs is not None else [default_media_steps()])
        self.active: list | None = None
        self._opened = 0
        self.spawned_argvs: list[list[str]] = []

    def next_job(self) -> list:
        if not self.jobs:
            raise ScriptExhausted("no more media jobs scripted")
        self.active = self.jobs.popleft()
        self._opened = 0
        return self.active

    def step(self) -> None:
        gates = _gates(self.active or [])
        if self._opened >= len(gates):
            raise ScriptExhausted("no more gates in active media job")
        gates[self._opened].open()
        self._opened += 1


class MemoryScript:
    """FakeMemoryReader 的快照队列：队首先出，只剩一个时重复返回（设计 §2.4）。"""

    def __init__(self, snapshots: list | None = None):
        self._snaps = deque(snapshots or [])
        self._lock = threading.Lock()

    def push(self, snapshot) -> None:
        with self._lock:
            self._snaps.append(snapshot)

    def current(self):
        with self._lock:
            if not self._snaps:
                raise ScriptExhausted("memory script is empty")
            if len(self._snaps) > 1:
                return self._snaps.popleft()
            return self._snaps[0]


class DownloadControl:
    """FakeDownloadExecutor 的控制面 + spawn 记录（设计 §2.4 DownloadControl）。

    「下载进度」= 测试向目的树写字节，真 Downloader 的目录采样如实读出。
    """

    def __init__(self):
        self.spawns: list[list[str]] = []
        self.handles: list = []

    @property
    def handle(self):
        if not self.handles:
            raise ScriptExhausted("no download spawned yet")
        return self.handles[-1]

    @staticmethod
    def dest_dir(argv: list[str]) -> Path:
        return Path(argv[argv.index("--local-dir") + 1])

    def advance(self, rel_path: str, n_bytes: int) -> None:
        dest = self.dest_dir(self.spawns[-1]) / rel_path
        dest.parent.mkdir(parents=True, exist_ok=True)
        dest.write_bytes(b"x" * n_bytes)

    def finish(self, manifest_files: list[tuple[str, int]]) -> None:
        for rel, size in manifest_files:
            self.advance(rel, size)
        self.handle.exit(0)

    def exit_terminated(self) -> None:
        h = self.handle
        assert h.terminated, "handle has not received terminate yet"
        h.exit(-15)
```

```python
class FakeLlmBackend:
    """LlmBackend 协议的假实现：spawn 即活、health 即真、chat_stream 走剧本。"""

    def __init__(self, script: ChatScript):
        self.script = script
        self.proc: _FakeBackendProcess | None = None

    def spawn(self, python, model_path, port, log_path):
        log_path = Path(log_path)
        log_path.parent.mkdir(parents=True, exist_ok=True)
        with open(log_path, "a", encoding="utf-8") as f:
            f.write("[fake mlx-lm] serving for e2e\n")
        self.proc = _FakeBackendProcess()
        return self.proc

    def health(self, port: int) -> bool:
        return self.proc is not None and self.proc.poll() is None

    def _iter_steps(self, payload: dict, *, wait_gates: bool):
        s = self.script
        if s.consumed:
            raise ScriptExhausted("chat script already consumed")
        s.consumed = True
        s.requests.append(payload)
        for step in s.steps:
            if isinstance(step, Gate):
                if wait_gates:
                    step.wait()
            elif isinstance(step, (Delta, Done)):
                yield step
            else:
                raise ScriptExhausted(f"unexpected chat step: {step!r}")

    def chat_stream(self, port: int, payload: dict):
        for step in self._iter_steps(payload, wait_gates=True):
            if isinstance(step, Delta):
                delta = {}
                if step.content is not None:
                    delta["content"] = step.content
                if step.reasoning is not None:
                    delta["reasoning_content"] = step.reasoning
                yield {"choices": [{"delta": delta, "finish_reason": None}]}
            else:
                yield {"choices": [{"delta": {}, "finish_reason": step.finish_reason}],
                       "usage": step.usage}

    def chat(self, port: int, payload: dict) -> dict:
        content, reasoning, done = [], [], None
        for step in self._iter_steps(payload, wait_gates=False):
            if isinstance(step, Delta):
                if step.content:
                    content.append(step.content)
                if step.reasoning:
                    reasoning.append(step.reasoning)
            else:
                done = step
        if done is None:
            raise ScriptExhausted("chat script has no Done step")
        message = {"role": "assistant", "content": "".join(content)}
        if reasoning:
            message["reasoning_content"] = "".join(reasoning)
        return {"choices": [{"message": message, "finish_reason": done.finish_reason}],
                "usage": done.usage}

    def log_tail(self, log_path: Path, max_lines: int = 40) -> str:
        log_path = Path(log_path)
        if not log_path.exists():
            return f"日志文件不存在: {log_path}"

    """media Executor 协议的 ProcessHandle 假实现，按单个作业剧本演进。"""

    def __init__(self, steps: list, output_path: Path):

        self._output_path = output_path

        self._cancel = __import__("threading").Event()
        self.terminated = False

    def iter_output(self):
        for step in self._steps:
            if isinstance(step, Gate):
                step.wait()
            elif isinstance(step, Line):
                yield step.text

            raise ScriptStall("wait() called before iter_output() finished")

        return self._code

        self._cancel.set()
```

```python
@dataclass
class TestHarness:
    base_url: str
    data_root: Path
    models_root: Path
    outputs_root: Path
    gateway_port: int
    stamp: str                      # FixedClock 派生的媒体产物 stamp
    chat: ChatScript
    media: MediaScript
    download: DownloadControl
    memory: MemoryScript
    seeded: dict[str, str]
    routes: list                    # [(method, path_prefix)] —— T-06 防漂移断言用
    _closers: list = field(default_factory=list, repr=False)

    def close(self) -> None:
        errors: list[Exception] = []
        closers, self._closers = list(self._closers), []
        for fn in reversed(closers):
            try:
                fn()
            except Exception as exc:            # noqa: BLE001 —— 收集后如实上抛
                errors.append(exc)
        if errors:
            raise RuntimeError(f"harness close failures: {errors!r}")

    def __enter__(self) -> "TestHarness":
        return self

    def __exit__(self, *exc) -> None:
        self.close()
```

```python
"""R-e2e-12：加载→完整流式聊天→卸载 的组合旅程中，前端对 :8767 的请求计数 == 0。"""
from playwright.sync_api import expect

from desk.testing import launch_test_harness


def test_full_journey_makes_zero_requests_to_8767(page, context, tmp_path, wait_until):
    with launch_test_harness(tmp_path) as h:
        page.goto(h.base_url)

        # 加载模型
        key = next(k for k, s in h.seeded.items()
                   if s == "present" and k not in ("h3", "music3"))
        page.get_by_role("combobox").first.select_option(key)
        page.get_by_role("button", name="加载").click()
        expect(page.get_by_text("卸载").first).to_be_visible()

        # 完整流式聊天（预开全部闸门，一次快进）
        h.chat.run_to_end()
        page.get_by_role("textbox").last.fill("零直连旅程")
        page.get_by_role("button", name="发送").click()
        expect(page.get_by_text("你好，世界").first).to_be_visible()

        # 卸载
        page.get_by_role("button", name="卸载").click()
        expect(page.get_by_role("button", name="加载")).to_be_visible()

        # 记录器真的在工作：页面资源 + 轮询 + SSE，必然远多于 10 条
        assert len(context.request_log) > 10, context.request_log
        # 旅程级零直连（专项断言；conftest 护栏 2 在 teardown 还会再断言一次）
        direct = [u for u in context.request_log if ":8767" in u]
        assert direct == [], direct
```
