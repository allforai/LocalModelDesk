```js
// 字节 / 百分比 / 时长 / 速率 / 时间戳 格式化。零 DOM、零 fetch。
const GB = 1024 ** 3;
const MB = 1024 ** 2;
const KB = 1024;

export function formatBytes(bytes) {
  if (!Number.isFinite(bytes) || bytes < 0) return "—";
  if (bytes >= GB) return `${(bytes / GB).toFixed(1)} GB`;
  if (bytes >= MB) return `${Math.round(bytes / MB)} MB`;
  if (bytes >= KB) return `${Math.round(bytes / KB)} KB`;
  return `${bytes} B`;
}

export function formatPercent(pct) {
  if (!Number.isFinite(pct)) return "0%";
  return `${Math.max(0, Math.min(100, Math.round(pct)))}%`;
}

export function formatDuration(seconds) {
  if (!Number.isFinite(seconds) || seconds < 0) return "—";
  const s = Math.round(seconds);
  if (s < 60) return `${s}秒`;
  const m = Math.floor(s / 60);
  if (m < 60) return `${m}分${String(s % 60).padStart(2, "0")}秒`;
  return `${Math.floor(m / 60)}时${String(m % 60).padStart(2, "0")}分`;
}

export function formatRate(bps) {
  if (!Number.isFinite(bps) || bps <= 0) return "—";
  return `${formatBytes(bps)}/s`;
}

export function formatTimestamp(iso) {
  if (typeof iso !== "string" || iso.length < 16) return "—";
  return `${iso.slice(0, 10)} ${iso.slice(11, 16)}`;
}
```

```js
// api:chatStream 的 SSE data 行 → {content, reasoning, done, error} 增量归并。
// 帧合同：llm 设计 routes.py 的 ChatEvent；容忍 [DONE] 哨兵与裸 {error} 信封。
export function initialStream() {
  return { content: "", reasoning: "", done: false, error: null, usage: null, finishReason: null };
}

export function reduceChunk(state, line) {
  if (state.done || state.error) return state;
  const raw = String(line).trim();
  if (raw === "") return state;
  if (raw === "[DONE]") return { ...state, done: true };
  let ev;
  try { ev = JSON.parse(raw); } catch { return state; }   // 坏行跳过，不中断
  if (ev === null || typeof ev !== "object") return state;
  if (ev.type === "done") {
    return { ...state, done: true, usage: ev.usage ?? null, finishReason: ev.finish_reason ?? null };
  }
  if (ev.type === "error") {
    return { ...state, error: { code: ev.code ?? "stream_error", message: ev.message ?? "流式对话出错" } };
  }
  if (ev.error !== undefined && ev.error !== null) {
    const e = typeof ev.error === "object" ? ev.error : { message: String(ev.error) };
    return { ...state, error: { code: e.code ?? "stream_error", message: e.message ?? String(ev.error) } };
  }
  if (ev.type === "delta") {
    const text = typeof ev.text === "string" ? ev.text : "";
    const reasoning = typeof ev.reasoning === "string" ? ev.reasoning : "";
    if (!text && !reasoning) return state;
    return { ...state, content: state.content + text, reasoning: state.reasoning + reasoning };
  }
  return state;
}
```

```js
// data:chatSession 列表的排序与展示标题推导（设计 §2.3 会话列表）。
const DEFAULT_TITLE = "新会话";

export function sortSessions(sessions) {
  return [...sessions].sort((a, b) =>
    String(b.updated ?? "").localeCompare(String(a.updated ?? "")));
}

export function displayTitle(session) {
  const title = (session.title ?? "").trim();
  if (title && title !== DEFAULT_TITLE) return title;
  const firstUser = (session.messages ?? []).find(
    (m) => m && m.role === "user" && typeof m.content === "string" && m.content.trim());
  if (!firstUser) return DEFAULT_TITLE;
  const text = firstUser.content.trim().replace(/\s+/g, " ");
  return text.length > 24 ? `${text.slice(0, 24)}…` : text;
}
```

```js
// data:modelStatus + data:modelEntry (+ data:downloadProgress) → 资源面板行视图（R-ui-05）。
import { formatBytes, formatPercent } from "./format.js";

export function rowView(entry, status, download = null) {
  const dl = download && download.state === "running" ? download : null;
  const isDownloading = dl !== null && dl.key === entry.key;
  const otherDownloading = dl !== null && dl.key !== entry.key;

  let badge;
  let pct = 0;
  if (status.state === "present") { badge = "齐"; pct = 100; }
  else if (status.state === "partial") { pct = status.percent; badge = `一半 ${formatPercent(status.percent)}`; }
  else if (status.state === "missing") { badge = "没下"; }
  else badge = `未知（${status.reason ?? "manifest_unavailable"}）`;

  let actions;
  if (isDownloading) actions = ["cancel"];
  else if (status.state === "present") actions = ["delete"];
  else if (status.state === "partial") actions = ["resume", "delete"];
  else if (status.state === "missing") actions = ["download"];
  else actions = [];

  const wantsDownload = actions.includes("download") || actions.includes("resume");
  const downloadDisabled = otherDownloading && wantsDownload;
  return {
    key: entry.key,
    badge,
    pct,
    sizeText: `${entry.gb} GB`,
    diskText: formatBytes(status.disk_bytes),
    actions,
    downloadDisabled,
    downloadDisabledReason: downloadDisabled ? "已有一个下载在进行（同时只允许一个）" : null,
    missing: status.gaps ?? [],
  };
}
```

```js
// data:deskState + data:memorySnapshot → 状态条视图（R-ui-07）。不含任何内存常量。
const GB = 1024 ** 3;
const REASON_TEXT = {
  media_busy: "媒体作业进行中",
  llm_loaded: "聊天模型驻留中",
  transition_in_progress: "重活交接进行中",
  memory_low: "可用内存不足",
};

export function renderState(deskState, snapshot) {
  const gb = (n) => (n / GB).toFixed(1);
  const memText = snapshot && Number.isFinite(snapshot.total_bytes)
    ? `已用 ${gb(snapshot.used_bytes)} / 总 ${gb(snapshot.total_bytes)} GB（可用 ${gb(snapshot.available_bytes)} GB）`
    : "内存读数不可用";

  const holder = deskState?.holder ?? null;
  let holderText = "空闲";
  if (holder) {
    if (holder.kind === "llm") holderText = `LLM：${holder.label}`;
    else if (holder.kind === "video") holderText = "视频生成中";
    else if (holder.kind === "music") holderText = "音乐生成中";
    else holderText = `${holder.kind}：${holder.label}`;
  }

  const mediaText = deskState?.media_busy ? "媒体：生成中" : "媒体：空闲";
  const can = deskState?.can_start ?? null;
```

```js
import { formatBytes } from "./format.js";

export function needsWarning(modelBytes, snapshot) {
  if (!Number.isFinite(modelBytes) || modelBytes <= 0) return { warn: false, message: null };
  const available = snapshot?.available_bytes;
  if (!Number.isFinite(available)) return { warn: false, message: null };
  if (modelBytes >= available) {

      message: `该模型约需 ${formatBytes(modelBytes)}，当前可用内存仅 ${formatBytes(available)}，` +
        `加载可能打满内存。确定继续？`,
    };
  }
  return { warn: false, message: null };
}
```

```python
"""ui 模块唯一的 Python 面：desk/static/ 的静态文件供给。

static_root 由服务组装层用 api:resolvePaths 解析出的 bundle 资源根传入
（<resources_root>/desk/static），本模块不拼任何路径（设计 §1.3）。
"""
from __future__ import annotations

from pathlib import Path
from urllib.parse import unquote

CONTENT_TYPES = {
    ".html": "text/html; charset=utf-8",
    ".css": "text/css; charset=utf-8",
    ".js": "text/javascript; charset=utf-8",
}


class StaticAssets:
    def __init__(self, static_root: Path) -> None:
        self._root = Path(static_root).resolve()

    def resolve(self, url_path: str) -> tuple[Path, str] | None:
        path = unquote(url_path.split("?", 1)[0])
        if path in ("", "/", "/index.html"):
            rel = "index.html"
        elif path.startswith("/static/"):
            rel = path[len("/static/"):]
        else:
            return None
        if not rel or rel.endswith("/"):
            return None
        candidate = (self._root / rel).resolve()
        if not candidate.is_relative_to(self._root):
            return None  # 路径逃逸 → 404（设计 §4）
        ctype = CONTENT_TYPES.get(candidate.suffix)
        if ctype is None:
            return None  # 表外后缀不供给
        if not candidate.is_file():
            return None
        return candidate, ctype
```

```js
function streamOf(chunks) {
  const encoder = new TextEncoder();
  return new ReadableStream({
    start(controller) {
      for (const c of chunks) controller.enqueue(typeof c === "string" ? encoder.encode(c) : c);
      controller.close();
    },
  });

async function collect(stream) {
  const out = [];
  for await (const line of sseDataLines(stream)) out.push(line);
  return out;
}
```

```js
// fetch ReadableStream → SSE data 行载荷。薄、DOM 无关（设计 §1.1）。
export async function* sseDataLines(stream) {
  const reader = stream.getReader();
  const decoder = new TextDecoder();
  let buf = "";
  const payload = (line) => line.slice(5).trimStart();
  try {
    for (;;) {
      const { done, value } = await reader.read();
      if (done) break;
      buf += decoder.decode(value, { stream: true });
      let idx;
      while ((idx = buf.indexOf("\n")) !== -1) {
        const line = buf.slice(0, idx).replace(/\r$/, "");
        buf = buf.slice(idx + 1);
        if (line.startsWith("data:")) yield payload(line);
      }
    }
    buf += decoder.decode();
    const rest = buf.replace(/\r$/, "");
    if (rest.startsWith("data:")) yield payload(rest);
  } finally {
    reader.releaseLock();
  }
}
```

```js
// 极轻量状态容器：tick 最新快照 + 当前会话 id（设计 §3）。无持久化。
export function createStore(initial = {}) {
  let state = { ...initial };
  const subs = new Set();
  return {
    get: () => state,
    set(patch) {
      state = { ...state, ...patch };
      for (const fn of [...subs]) fn(state);
    },
    subscribe(fn) {
      subs.add(fn);
      return () => subs.delete(fn);
    },
  };
}
```

```js
// ui:chatPane —— 模型选择/加载/卸载、流式回复（思考可展开）、左侧会话列表（R-ui-02、R-ui-08）。
// 只做「取数 → 调纯函数 → 写 DOM」；一律 createElement/textContent，无 innerHTML。
import * as api from "../api.js";
import { sseDataLines } from "../stream.js";
import { initialStream, reduceChunk } from "../pure/chat_stream.js";
import { needsWarning } from "../pure/mem_warn.js";
import { sortSessions, displayTitle } from "../pure/sessions.js";
import { confirmDialog } from "../widgets/confirm.js";

export function createChatPane(root) {
  const doc = root.ownerDocument;
  const els = {
    sessionList: root.querySelector("[data-session-list]"),
    sessionNew: root.querySelector("[data-session-new]"),
    modelSelect: root.querySelector("[data-model-select]"),
    loadBtn: root.querySelector("[data-load]"),
    unloadBtn: root.querySelector("[data-unload]"),
    modelState: root.querySelector("[data-model-state]"),
    messages: root.querySelector("[data-messages]"),
    input: root.querySelector("[data-chat-input]"),
    sendBtn: root.querySelector("[data-send]"),
    error: root.querySelector("[data-chat-error]"),
  };
  let catalog = [];
  let sessions = [];
  let currentId = null;
  let streaming = false;

  const setError = (text) => { els.error.textContent = text ?? ""; };
  const current = () => sessions.find((s) => s.id === currentId) ?? null;

  // ---- 模型区 ----
  async function refreshModels() {
    const [entries, verified, status] = await Promise.all([
      api.listCatalog(), api.verifyAllModels(), api.llmStatus(),
    ]);
    catalog = entries.filter((e) => e.group === "chat");
    const byKey = new Map(verified.models.map((m) => [m.key, m]));
    const selected = els.modelSelect.value;
    els.modelSelect.replaceChildren();
    for (const entry of catalog) {
      const ready = byKey.get(entry.key)?.state === "present";
      const opt = doc.createElement("option");
      opt.value = entry.key;
      const marks = [entry.params, entry.quant, entry.vision ? "视觉" : null].filter(Boolean).join(" · ");
      opt.textContent =
        `${entry.name} · ${entry.gb} GB${marks ? ` · ${marks}` : ""}${ready ? "" : "（未下载/不完整）"}`;
      opt.disabled = !ready;
      els.modelSelect.append(opt);
    }
    if (selected) els.modelSelect.value = selected;
    renderLlm(status);
  }

  function renderLlm(payload) {           // payload = {state: data:llmState, loaded_model?}
    const s = payload.state;
    els.modelState.dataset.status = s.status;
    if (s.status === "idle") els.modelState.textContent = "未加载";
    else if (s.status === "loading") els.modelState.textContent = `加载中：${s.model_key ?? ""}`;
    else if (s.status === "loaded") {
      els.modelState.textContent = `已加载：${payload.loaded_model?.name ?? s.model_key}`;
      if (s.model_key) els.modelSelect.value = s.model_key;   // 当前加载模型高亮
    } else {
      const tail = s.error?.log_tail ? `\n${s.error.log_tail}` : "";
      els.modelState.textContent =
        `加载失败（${s.error?.code ?? "?"}）：${s.error?.message ?? ""}${tail}`;   // R-llm-04 原样
    }
  }

  async function loadSelected() {
    setError("");
    const entry = catalog.find((e) => e.key === els.modelSelect.value);
    if (!entry) return;
    try {
      const snapshot = await api.memorySnapshot();                       // R-ui-08 第 1 步
      const { warn, message } = needsWarning(Math.round(entry.gb * 1024 ** 3), snapshot);
      if (warn) {
        const go = await confirmDialog(doc, { title: "内存警告", message, confirmLabel: "仍要加载" });
        if (!go) return;                                                 // 取消即中止
      }
      await api.loadLlm(entry.key);
      await pollUntilSettled();
    } catch (err) { setError(err.message); }
  }

  async function pollUntilSettled() {
    for (;;) {
      const payload = await api.llmStatus();
      renderLlm(payload);
      if (payload.state.status !== "loading") return;
      await new Promise((r) => setTimeout(r, 1000));
    }
  }

  async function unload() {
    setError("");
    try {
      await api.unloadLlm();
      renderLlm(await api.llmStatus());
    } catch (err) { setError(err.message); }
  }

  // ---- 会话列表 ----
  async function refreshSessions(selectId) {
    sessions = sortSessions(await api.listChatSessions());
    if (sessions.length === 0) sessions = [await api.createChatSession()];
    const wanted = selectId ?? currentId;
    currentId = sessions.some((s) => s.id === wanted) ? wanted : sessions[0].id;
    renderSessionList();
    renderMessages();
  }

  function renderSessionList() {
    els.sessionList.replaceChildren();
    for (const session of sessions) {
      const li = doc.createElement("li");
      li.classList.toggle("active", session.id === currentId);
      const title = doc.createElement("span");
      title.className = "session-title";
      title.textContent = displayTitle(session);
      title.addEventListener("click", () => {
        currentId = session.id;
        renderSessionList();
        renderMessages();
      });
      const rename = doc.createElement("button");
      rename.textContent = "改名";
      rename.addEventListener("click", () => beginRename(li, session));
      const del = doc.createElement("button");
      del.textContent = "删";
      del.addEventListener("click", async () => {
        const go = await confirmDialog(doc, {
          title: "删除会话",
          message: `确定删除「${displayTitle(session)}」？该会话的全部消息将被删除。`,
          confirmLabel: "删除",
        });
        if (!go) return;
        try {
          await api.deleteChatSession(session.id);
          await refreshSessions();
        } catch (err) { setError(err.message); }
      });
      li.append(title, rename, del);
      els.sessionList.append(li);
    }
  }

  function beginRename(li, session) {
    const input = doc.createElement("input");
    input.value = session.title ?? "";
    let done = false;
    const finish = async () => {
      if (done) return;
      done = true;
      const title = input.value.trim();
      if (title && title !== session.title) {
        try { await api.updateChatSession(session.id, { title }); }
        catch (err) { setError(err.message); }
      }
      await refreshSessions(session.id);
    };
    input.addEventListener("keydown", (e) => { if (e.key === "Enter") finish(); });
    input.addEventListener("blur", finish);
    li.replaceChildren(input);
    input.focus();
  }

  // ---- 消息流 ----
  function messageNode(msg, live = false) {
    const wrap = doc.createElement("div");
    wrap.className = `msg msg-${msg.role}`;
    const details = doc.createElement("details");
    details.className = "reasoning";
    const summary = doc.createElement("summary");
    summary.textContent = "思考过程";
    const reasoningEl = doc.createElement("pre");
    details.append(summary, reasoningEl);
    const contentEl = doc.createElement("div");
    contentEl.className = "msg-content";
    const errorEl = doc.createElement("p");
    errorEl.className = "inline-error";
    if (msg.role === "assistant") wrap.append(details);
    wrap.append(contentEl, errorEl);
    reasoningEl.textContent = msg.reasoning ?? "";
    contentEl.textContent = msg.content ?? "";
    details.hidden = !msg.reasoning && !live;
    details.open = live;                       // 生成中默认展开，有正文后折叠
    els.messages.append(wrap);
    els.messages.scrollTop = els.messages.scrollHeight;
    return { details, reasoningEl, contentEl, errorEl };
  }

  function renderMessages() {
    els.messages.replaceChildren();
    for (const msg of current()?.messages ?? []) messageNode(msg);
  }

  async function send() {
    if (streaming) return;
    const text = els.input.value.trim();
    const session = current();
    if (!text || !session) return;
    setError("");
    session.messages = session.messages ?? [];
    session.messages.push({ role: "user", content: text });
    els.input.value = "";
    messageNode({ role: "user", content: text });
    const live = messageNode({ role: "assistant", content: "", reasoning: "" }, true);
    streaming = true;
    els.sendBtn.disabled = true;
    let state = initialStream();
    try {
      const body = await api.chatStream(session.messages);
      for await (const line of sseDataLines(body)) {
        state = reduceChunk(state, line);
        live.reasoningEl.textContent = state.reasoning;
        live.details.hidden = !state.reasoning;
        if (state.content && live.details.open) live.details.open = false;
        live.contentEl.textContent = state.content;
        if (state.done || state.error) break;
      }
      if (!state.done && !state.error) {
        state = { ...state, error: { code: "stream_interrupted", message: "流在完成前中断" } };
      }
    } catch (err) {
      // 被拒（无模型/媒体在跑，R-llm-06）与网络错误：内联展示服务端原因，不空转圈
      state = { ...state, error: { code: err.code ?? "stream_error", message: err.message } };
    } finally {
      streaming = false;
      els.sendBtn.disabled = false;
    }
    if (state.error) {
      live.errorEl.textContent = `出错（${state.error.code}）：${state.error.message}`;
      session.messages.pop();   // 已收增量保留展示；本轮不写回会话（设计 §4）
      return;
    }
    const assistant = { role: "assistant", content: state.content };
    if (state.reasoning) assistant.reasoning = state.reasoning;   // A7 消息形状
    session.messages.push(assistant);
    try {
      const updated = await api.updateChatSession(session.id, {
        messages: session.messages,
        model: els.modelSelect.value || null,
      });
      sessions = sortSessions(sessions.map((s) => (s.id === updated.id ? updated : s)));
      renderSessionList();
    } catch (err) { setError(`会话保存失败：${err.message}`); }
  }

  els.loadBtn.addEventListener("click", loadSelected);
  els.unloadBtn.addEventListener("click", unload);
  els.sendBtn.addEventListener("click", send);
  els.input.addEventListener("keydown", (e) => {
    if (e.key === "Enter" && (e.metaKey || e.ctrlKey)) send();
  });
  els.sessionNew.addEventListener("click", async () => {
    try {
      const created = await api.createChatSession();
      await refreshSessions(created.id);
    } catch (err) { setError(err.message); }
  });

  async function init() {
    try { await refreshModels(); } catch (err) { setError(err.message); }
    try { await refreshSessions(); } catch (err) { setError(err.message); }
  }

  return { init, refreshModels };
}
```

```js
// 数据按「进入面板 + 操作后」拉取（A8）；下载进行中由 main tick 调 tickDownload()。
import * as api from "../api.js";
import { rowView } from "../pure/model_status.js";
import { formatBytes, formatRate, formatDuration } from "../pure/format.js";
import { confirmDialog } from "../widgets/confirm.js";

const ACTION_LABEL = { download: "下载", resume: "续传", delete: "删除", cancel: "取消" };

export function createResourcesPane(root) {
  const doc = root.ownerDocument;
  const els = {
    rootPath: root.querySelector("[data-res-root]"),
    usage: root.querySelector("[data-res-usage]"),
    refreshBtn: root.querySelector("[data-res-refresh]"),
    list: root.querySelector("[data-res-list]"),
    error: root.querySelector("[data-res-error]"),
  };
  let catalog = [];
  let statuses = [];
  let download = null;
  let watching = false;

  const setError = (t) => { els.error.textContent = t ?? ""; };

  async function refresh(reverify = false) {
    setError("");
    try {
      const [entries, verified, progress] = await Promise.all([
        api.listCatalog(), api.verifyAllModels(reverify), api.downloadProgress(),
      ]);
      catalog = entries;
      statuses = verified.models;
      download = progress;
      watching = progress.state === "running";
      renderHead(verified.disk ?? await api.diskUsage());
      renderRows();
    } catch (err) { setError(err.message); }
  }

  function renderHead(disk) {   // DiskUsage：models_root/free_bytes/total_bytes/per_model
    els.rootPath.textContent = disk.models_root;
    const used = Object.values(disk.per_model ?? {}).reduce((a, b) => a + b, 0);
    els.usage.textContent =
      `已占盘 ${formatBytes(used)} · 卷剩余 ${formatBytes(disk.free_bytes)}（总 ${formatBytes(disk.total_bytes)}）`;
  }

  function renderRows() {
    els.list.replaceChildren();
    const byKey = new Map(statuses.map((s) => [s.key, s]));
    for (const entry of catalog) {
      const status = byKey.get(entry.key);
      if (status) els.list.append(rowNode(entry, status));
    }
  }

  function rowNode(entry, status) {
    const view = rowView(entry, status, download);
    const li = doc.createElement("li");
    li.className = "res-row";
    const name = doc.createElement("span");
    name.className = "res-name";
    name.textContent = `${entry.name}${entry.vision ? " · 视觉" : ""}`;
    const badge = doc.createElement("span");
    badge.className = `badge badge-${status.state}`;
    badge.textContent = view.badge;
    const size = doc.createElement("span");
    size.textContent = `声明 ${view.sizeText} · 占盘 ${view.diskText}`;
    const actions = doc.createElement("span");
    actions.className = "res-actions";
    for (const action of view.actions) {
      const btn = doc.createElement("button");
      btn.textContent = ACTION_LABEL[action];
      if ((action === "download" || action === "resume") && view.downloadDisabled) {
        btn.disabled = true;
        btn.title = view.downloadDisabledReason;   // R-resources-09：注明原因
      }
      btn.addEventListener("click", () => runAction(action, entry, status));
      actions.append(btn);
    }
    const rowErr = doc.createElement("p");
    rowErr.className = "inline-error";
    rowErr.dataset.rowError = entry.key;
    li.append(name, badge, size, actions);
    if (download?.state === "running" && download.key === entry.key) li.append(progressNode());
    if (view.missing.length) li.append(gapsNode(view.missing));
    li.append(rowErr);
    return li;
  }

  function progressNode() {   // data:downloadProgress 直接格式化展示
    const box = doc.createElement("div");
    box.className = "dl-progress";
    const bar = doc.createElement("progress");
    bar.max = 100;
    bar.value = download.percent ?? 0;
    const text = doc.createElement("span");
    const eta = download.eta_seconds == null ? "—" : formatDuration(download.eta_seconds);
    text.textContent =
      `${formatBytes(download.bytes_done)} / ${formatBytes(download.bytes_total)}` +
      ` · ${formatRate(download.rate_bps)} · 剩余 ${eta}` +
      `${download.current_file ? ` · ${download.current_file}` : ""}`;
    box.append(bar, text);
    return box;
  }

  function gapsNode(gaps) {   // partial 可展开：缺哪些文件（R-ui-05）
    const details = doc.createElement("details");
    const summary = doc.createElement("summary");
    summary.textContent = `缺失/短缺 ${gaps.length} 个文件`;
    details.append(summary);
    const ul = doc.createElement("ul");
    for (const gap of gaps) {
      const li = doc.createElement("li");
      li.textContent = gap.local_size === 0
        ? `${gap.path} — 完全缺失（需 ${formatBytes(gap.expected_size)}）`
        : `${gap.path} — 短缺（${formatBytes(gap.local_size)} / ${formatBytes(gap.expected_size)}）`;
      ul.append(li);
    }
    details.append(ul);
    return details;
  }

  async function runAction(action, entry, status) {
    const rowErr = els.list.querySelector(`[data-row-error="${entry.key}"]`);
    if (rowErr) rowErr.textContent = "";
    try {
      if (action === "download" || action === "resume") {
        download = await api.startDownload(entry.key);   // 续传同走 startDownload（设计 §2.6）
        watching = true;
        renderRows();
      } else if (action === "cancel") {
        download = await api.cancelDownload();           // 已下部分保留，可续传
        await refresh();
      } else if (action === "delete") {
        const go = await confirmDialog(doc, {
          title: "删除模型",
          message: `确定删除「${entry.name}」？将回收约 ${formatBytes(status.disk_bytes)} 磁盘空间。`,
          confirmLabel: "删除",
        });
        if (!go) return;                                 // 取消则根本不发请求
        await api.deleteModel(entry.key);                // api.js 恒带 confirm=key
        await refresh();
      }
    } catch (err) {
      if (rowErr) rowErr.textContent = err.message;      // 下载被拒/失败：行内展示，进度保留
      else setError(err.message);
    }
  }

  async function tickDownload() {   // main tick 在面板可见且有下载时调用（A3）
    if (!watching) return;
    download = await api.downloadProgress();
    if (download.state === "running") renderRows();
    else { watching = false; await refresh(); }
  }

  els.refreshBtn.addEventListener("click", () => refresh(true));
  return { refresh, tickDownload };
}
```

```js
    } catch (err) { els.error.textContent = err.message; }

  function render(outputs, history) {
    els.list.replaceChildren();
    const outByName = new Map(outputs.map((o) => [o.name, o]));
    const rows = [];
    for (const entry of history) {
      rows.push({ ts: entry.ts ?? "", entry, output: entry.output ? outByName.get(entry.output) ?? null : null });
    }
    for (const o of outputs) {
      if (o.orphan) rows.push({ ts: o.ts ?? "", entry: null, output: o });   // 孤儿成品
    }
    rows.sort((a, b) => String(b.ts).localeCompare(String(a.ts)));
    for (const row of rows) els.list.append(rowNode(row));

      ? [p.width && p.height ? `${p.width}×${p.height}` : null,
         p.frames ? `${p.frames}帧` : null, p.steps ? `${p.steps}步` : null]
      : [p.duration ? `${p.duration}s` : null];
    return {
      text: text.length > 60 ? `${text.slice(0, 60)}…` : text,
      spec: spec.filter(Boolean).join(" · "),
    };
  }

  function rowNode({ entry, output }) {

    li.className = "lib-row";
    const head = doc.createElement("div");

        err.className = "inline-error";
        err.textContent = entry.error;
        head.append(err);
      }
```

```js
// ui:deskShell —— 启动序列、面板切换（hidden 切换不销毁）、单一 2s 轮询循环（设计 §2.1、A3、A4）。

let panes = null;

let settings = null;
let failures = 0;
let jobActive = false;
let jobLogFrom = 0;

async function boot() {
  let config;
  try { config = await api.readConfig(); }
  catch (err) { return fatal(`无法读取配置：${err.message}`); }
  if (config.needs_setup ?? !config.first_run_done) return enterFirstRun(config);
  enterDesk();
}

function fatal(message) {
  const el = $("#fatal");
  el.hidden = false;
  el.textContent = message;   // 错误就是错误：不渲染假台面
}

function enterFirstRun(config) {
  const el = $("#pane-firstrun");
  const pane = createFirstRunPane(el, { onDone: () => { el.hidden = true; boot(); } });
  pane.init(config);
  el.hidden = false;
}

function enterDesk() {
  if (!panes) {
    statusbar = createStatusBar($("#statusbar"));
    const onStarted = (job) => { jobActive = true; lastJobId = job.job_id; jobLogFrom = 0; };
    panes = {
      chat: createChatPane($("#pane-chat")),
      video: createVideoPane($("#pane-video"), { onStarted }),
      music: createMusicPane($("#pane-music"), { onStarted }),
      resources: createResourcesPane($("#pane-resources")),
      library: createLibraryPane($("#pane-library"), { applyFill }),
    };
    settings = createSettingsPane($("#pane-settings"));
    $("[data-open-settings]").addEventListener("click", () => {
      $("#pane-settings").hidden = false;
      settings.refresh();
    });
    $("[data-close-settings]").addEventListener("click", () => {
      $("#pane-settings").hidden = true;
    });
    for (const btn of document.querySelectorAll("#tabs [data-tab]")) {
      btn.addEventListener("click", () => showTab(btn.dataset.tab));
    }
    setInterval(tick, 2000);   // 唯一定时器（A3）
  }
  panes.chat.init();
  tick();
}

function showTab(name) {
  for (const btn of document.querySelectorAll("#tabs [data-tab]")) {
    btn.classList.toggle("active", btn.dataset.tab === name);
  }
  for (const section of document.querySelectorAll("main > [data-pane]")) {
    section.hidden = section.dataset.pane !== name;   // hidden 切换，不销毁（A4）
  }
  store.set({ activeTab: name });
  if (name === "resources") panes.resources.refresh();   // A8：进入面板时拉取
  if (name === "library") panes.library.refresh();
}

function applyFill(plan) {   // 跨面板动作经 main（设计 §3）
  if (!plan) return;
  panes[plan.pane].fill(plan.fields);
  showTab(plan.pane);
}

async function tick() {
  try {
    const [deskState, memory] = await Promise.all([api.deskState(), api.memorySnapshot()]);
    failures = 0;
    statusbar.offline(false);
    statusbar.update(deskState, memory);
    store.set({ deskState, memory });
    if (deskState.media_busy) jobActive = true;
    if (jobActive) await tickJob();
    if (store.get().activeTab === "resources") await panes.resources.tickDownload();
  } catch (err) {
    failures += 1;                                  // 单次失败保持旧数据
    if (failures >= 3) statusbar.offline(true);     // 连续 ≥3 次 → 服务失联
  }
}

async function tickJob() {
  const payload = await api.jobStatus(jobLogFrom, lastJobId);
  const pane = payload.kind === "music" ? panes.music : panes.video;
  if (payload.job_id !== lastJobId) {   // 新作业（或页面刷新后首见）：重置渲染与游标
    pane.jobView.reset();
    lastJobId = payload.job_id;
  }
  pane.jobView.apply(payload);
  jobLogFrom = payload.next_log_from ?? jobLogFrom;
  if (payload.status !== "running") jobActive = false;
}

boot();
```
